#!/usr/bin/env python3
import pathlib

def gather_files_content(output_file: pathlib.Path, extensions: tuple) -> None:
    """
    Recursively gather the content of files with specified extensions in the current directory 
    and its subdirectories and save it to a text file.

    This function searches for files that have an extension matching any in the provided tuple.
    It writes a header with the file's relative path and then writes the file's content 
    to the output file. Each file's content is separated by a line of equal signs.
    
    Args:
        output_file (pathlib.Path): The file path to which the output should be written.
        extensions (tuple): A tuple of file extensions to search for (e.g., (".py", ".html")).
    
    Returns:
        None
    """
    current_path = pathlib.Path(".")
    with output_file.open("w", encoding="utf-8") as out_file:
        for file_path in current_path.rglob("*"):
            if file_path.suffix in extensions:
                out_file.write(f"File: {file_path}\n\n")
                try:
                    with file_path.open("r", encoding="utf-8") as file:
                        content = file.read()
                        out_file.write(content)
                except Exception as e:
                    out_file.write(f"Error reading file {file_path}: {e}")
                out_file.write("\n" + "=" * 80 + "\n\n")

def main() -> None:
    """
    The main entry point of the script.

    Creates an output text file 'files_dump.txt' in the current directory and writes the gathered
    content of all Python and HTML files to it.
    """
    output_path = pathlib.Path("files_dump.txt")
    gather_files_content(output_path, (".py", ".html"))
    print(f"Dump saved to {output_path}")

if __name__ == "__main__":
    main()

def parse_datetime_string(datetime_str):
    """Attempts to parse various datetime string formats into timezone-aware UTC datetime objects."""
    if not datetime_str:
        return None
    try:
        # Prioritize ISO format parsing
        dt = datetime.fromisoformat(datetime_str.replace('Z', '+00:00')) # Handle Z for UTC
        # If offset-naive, assume UTC
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=timezone.utc)
        # Convert to UTC if it has other timezone info
        return dt.astimezone(timezone.utc)
    except ValueError:
        # Fallback using dateutil if available and ISO parsing failed
        if HAS_DATEUTIL:
            try:
                # fuzzy=False requires a closer match
                dt = dateutil_parser.parse(datetime_str, fuzzy=False)
                if dt.tzinfo is None:
                     dt = dt.replace(tzinfo=timezone.utc) # Assume UTC if naive
                return dt.astimezone(timezone.utc) # Convert to UTC
            except (ValueError, OverflowError, TypeError) as e:
                print(f"dateutil parsing failed for '{datetime_str}': {e}")
                return None
        else:
             print(f"ISO parsing failed for '{datetime_str}' and dateutil not available.")
             return None
    except Exception as e: # Catch other potential errors
        print(f"Generic parsing error for '{datetime_str}': {e}")
        return None

def call_openai_llm(student_data, user_name="Student"):
    """Calls OpenAI API to generate event proposals, requesting specific datetime output."""
    print(f"--- Calling OpenAI API for user: {user_name} ---")
    print(f"Raw Input Data: {student_data}")
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key: raise ValueError("OpenAI API key not found in environment variables.")
    try: client = OpenAI(api_key=api_key)
    except Exception as e: raise ValueError(f"Failed to initialize OpenAI client: {e}")

    # Get current time in UTC for the prompt
    # You might adjust this to user's local time if needed, but UTC is generally better for backend consistency
    now_utc = datetime.now(timezone.utc)
    current_time_str = now_utc.strftime('%Y-%m-%dT%H:%M:%SZ') # ISO 8601 format

    # FIXED: Updated system prompt to request specific datetime format
    system_prompt = """
    You are an assistant for Irvine Valley College (IVC) students helping them organize study groups and social meetups.
    The current time is {current_time}. Analyze the user's desired meetup (classes, purpose, availability) and propose potential meetup events.
    Focus on the classes/purpose mentioned. Suggest plausible IVC locations.
    Based on the user's availability and the current time, determine a specific future date and time for the event.

    You MUST output your response as a JSON list of objects. Each object represents one proposal and MUST have keys:
    - "type": String (e.g., "Study Group", "Social Hangout").
    - "subject": String (e.g., "CIS 1A Lab 3 Session").
    - "details": String, brief description including who initiated it ({user_name}).
    - "proposed_datetime": String, **A specific future date and time for the event in ISO 8601 format (YYYY-MM-DDTHH:MM:SSZ). Calculate this based on the current time and user input.**
    - "proposed_location": String (e.g., "Library Study Room 3B", "Online/Zoom").
    - "reasoning": String, brief explanation for the suggestion.

    Example User Input: {{"classes": "MATH 1A", "availability": "Next Tuesday afternoon", "purpose": "Review derivatives"}}
    Example JSON output (assuming current time is 2025-04-04T23:30:00Z):
    [
      {{
        "type": "Study Group",
        "subject": "MATH 1A Derivatives Review",
        "details": "Review session for MATH 1A derivatives, initiated by {user_name}.",
        "proposed_datetime": "2025-04-08T14:00:00Z", // Example: Calculated Next Tuesday 2 PM UTC
        "proposed_location": "IVC Math Center (LRC)",
        "reasoning": "Matches class, purpose, and availability. Proposing next Tuesday afternoon UTC."
      }}
    ]
    Generate 1-2 relevant suggestions. Ensure the entire output is ONLY the JSON list.
    """.format(user_name=user_name, current_time=current_time_str) # Pass current time to prompt

    user_prompt = f"Here is the student's desired meetup information:\n{json.dumps(student_data)}"

    try:
        print(f"Sending request to OpenAI API (Current Time: {current_time_str})...")
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            response_format={ "type": "json_object" }
        )
        print("Received response from OpenAI API.")
        json_string = response.choices[0].message.content
        print(f"Raw API Response Content:\n{json_string}")
        parsed_response = json.loads(json_string)

        # Handle potentially nested JSON list
        if isinstance(parsed_response, list): petitions_list = parsed_response
        elif isinstance(parsed_response, dict) and len(parsed_response) > 0:
            potential_list = next((v for v in parsed_response.values() if isinstance(v, list)), None)
            if potential_list is not None: petitions_list = potential_list
            else: raise ValueError("JSON response is dict, but no list found.")
        else: raise ValueError("Unexpected JSON structure from API.")

        # FIXED: Validate for the new 'proposed_datetime' key
        valid_petitions = []
        required_keys = ["type", "subject", "details", "proposed_datetime", "proposed_location", "reasoning"]
        for item in petitions_list:
            if isinstance(item, dict) and all(key in item for key in required_keys):
                valid_petitions.append(item)
            else:
                print(f"Warning: Proposal object missing required keys or not a dict: {item}")

        print(f"--- OpenAI Call Successful. Valid Proposals: {json.dumps(valid_petitions, indent=2)} ---")
        return valid_petitions

    except Exception as e:
        print(f"ERROR: OpenAI API call or processing failed: {e}")
        raise

@app.route('/')
def index():
    """Displays existing events with filtering and sorting."""
    if not current_user.is_authenticated:
         flash('Please log in or sign up.', 'info'); return redirect(url_for('login'))

    subject_filter = request.args.get('subject', '').strip()
    # FIXED: Added 'event_time_asc' sorting option
    sort_by = request.args.get('sort_by', 'event_time_asc') # Default sort: Soonest event first

    try:
        query = Event.query.options(
            db.joinedload(Event.creator),
            db.subqueryload(Event.comments).joinedload(Comment.author),
            db.subqueryload(Event.commitments).joinedload(Commitment.user)
        )

        if subject_filter:
            query = query.filter(Event.subject.ilike(f'%{subject_filter}%'))

        # Apply Sorting
        if sort_by == 'date_asc':
            query = query.order_by(Event.created_at.asc())
        elif sort_by == 'date_desc':
             query = query.order_by(Event.created_at.desc())
        elif sort_by == 'event_time_asc':
            # Sort by the actual proposed event time (soonest first)
            # Put events with no parsable time at the end
            query = query.order_by(nullslast(Event.proposed_datetime.asc()))
        else: # Default to event_time_asc if invalid sort_by is given
             query = query.order_by(nullslast(Event.proposed_datetime.asc()))

        events = query.all()

    except Exception as e:
        print(f"Error fetching events: {e}"); flash("Error loading events.", "error"); events = []

    current_filters = { 'subject': subject_filter, 'sort_by': sort_by }
    return render_template('index.html', events=events, filters=current_filters)

def propose_event_form():
    """Handles event proposal, parses datetime, saves to DB."""
    if request.method == 'POST':
        classes = request.form.get('classes', '').strip()
        availability = request.form.get('availability', '').strip()
        purpose = request.form.get('purpose', '').strip()
        if not availability or not purpose:
             flash('Purpose and availability are required.', 'error')
             return render_template('propose_form.html', form_data=request.form)

        student_input_data = { "classes": classes, "availability": availability, "purpose": purpose }
        try:
            user_identifier = current_user.name or current_user.email
            proposed_events_data = call_openai_llm(student_input_data, user_identifier)

            if not proposed_events_data:
                flash('AI could not generate suggestions.', 'info')
                return render_template('propose_form.html', form_data=request.form)

            num_saved = 0
            for event_data in proposed_events_data:
                # FIXED: Parse datetime string from LLM
                proposed_dt_str = event_data.get('proposed_datetime')
                parsed_datetime = parse_datetime_string(proposed_dt_str)
                if not parsed_datetime:
                     flash(f"Warning: Could not parse proposed time '{proposed_dt_str}' for event '{event_data.get('subject')}'. It won't be sortable by time.", "warning")

                try:
                    new_event = Event(
                        user_id=current_user.id,
                        subject=event_data.get('subject'),
                        type=event_data.get('type'),
                        # Store both the string and the parsed datetime (or None)
                        proposed_time_string=proposed_dt_str, # Store original string
                        proposed_datetime=parsed_datetime,     # Store parsed object (or None)
                        proposed_location=event_data.get('proposed_location'),
                        details=event_data.get('details'),
                        reasoning=event_data.get('reasoning'),
                        status='proposed'
                    )
                    db.session.add(new_event); num_saved += 1
                except Exception as e: print(f"Error creating event obj: {e}"); flash("Warning: Couldn't save AI suggestion.", "warning")

            if num_saved > 0:
                db.session.commit(); flash(f'{num_saved} new proposal(s) created!', 'success')
                return redirect(url_for('index'))
            else: flash('Could not save any AI suggestions.', 'error'); return render_template('propose_form.html', form_data=request.form)

        except ValueError as ve: flash(f'Config Error: {ve}', 'error')
        except Exception as e: db.session.rollback(); flash(f'Unexpected error: {e}', 'error'); print(f"Error: {e}")
        return render_template('propose_form.html', form_data=request.form)

    return render_template('propose_form.html')

<script>
    document.addEventListener('DOMContentLoaded', () => {
        const eventList = document.querySelector('.event-list');
        if (eventList) {
            eventList.addEventListener('click', async (event) => {
                if (!event.target.classList.contains('commit-btn')) { return; }
                const button = event.target; const eventId = button.getAttribute('data-event-id');
                const eventCard = button.closest('.event-card'); const feedbackDiv = eventCard.querySelector(`#feedback-${eventId}`);
                const committedListSpan = eventCard.querySelector(`#committed-list-${eventId}`); button.disabled = true;
                feedbackDiv.textContent = 'Processing...'; feedbackDiv.style.color = '#555';
                try {
                    const response = await fetch(`/commit/${eventId}`, { method: 'POST', headers: {'Accept': 'application/json'} });
                    const result = await response.json();
                    if (response.ok && response.status === 201) {
                        feedbackDiv.textContent = 'Committed!'; feedbackDiv.style.color = 'green'; button.textContent = 'Committed';
                        button.classList.remove('commit-btn'); button.classList.add('committed-btn');
                        const userName = result.user_name || 'You';
                        if (committedListSpan) {
                            const currentText = committedListSpan.textContent.trim(); const strongTag = committedListSpan.previousElementSibling;
                            let currentCount = 0; if (strongTag && strongTag.textContent.includes('(')) { currentCount = parseInt(strongTag.textContent.match(/\d+/)[0] || 0); }
                            if (currentText.includes('No one yet')) { committedListSpan.textContent = userName; } else { committedListSpan.textContent += `, ${userName}`; }
                            if (strongTag) { strongTag.textContent = `Committed (${currentCount + 1}):`; }
                        }
                    } else { feedbackDiv.textContent = `Error: ${result.error || 'Could not commit'}`; feedbackDiv.style.color = 'red'; button.disabled = false; }
                } catch (error) { console.error('Commit AJAX error:', error); feedbackDiv.textContent = 'Network error.'; feedbackDiv.style.color = 'red'; button.disabled = false; }
            });
        }
    });
</script>
